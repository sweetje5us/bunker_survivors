🐛 Баг 2: Застревание при переходе в комнаты после 5-й

Проблема: Жители застревали справа при попытке перехода в следующую комнату, если комнат в ряду больше 5.
Решение: Полностью переписана система построения пути для комнат на одном этаже.
// Улучшенная функция определения соседних комнат с увеличенным допуском const roomsAreAdjacent = (a: Phaser.Geom.Rectangle, b: Phaser.Geom.Rectangle): boolean => { if (Math.abs(a.y - b.y) > 1e-3) return false // Увеличиваем допуск для комнат, которые могут быть далеко друг от друга const rightToLeft = Math.abs((a.x + a.width) - b.x) <= GAP_TOL + 20 const leftToRight = Math.abs((b.x + b.width) - a.x) <= GAP_TOL + 20 return rightToLeft || leftToRight } // Новая функция построения пути - создает прямой путь через все комнаты на этаже const buildHorizontalChain = (fromIdx: number, toIdx: number): number[] | null => { // Находим все комнаты на том же этаже const roomsOnSameFloor = [] for (let i = 0; i < this.roomRects.length; i++) { if (Math.abs(this.roomRects[i].y - sameY) < 1e-3) { roomsOnSameFloor.push(i) } } // Сортируем комнаты по X координате roomsOnSameFloor.sort((a, b) => this.roomRects[a].x - this.roomRects[b].x) // Создаем путь через все комнаты между fromIdx и toIdx const path: number[] = [] const start = Math.min(fromPos, toPos) const end = Math.max(fromPos, toPos) for (let i = start; i <= end; i++) { path.push(roomsOnSameFloor[i]) } return path }
